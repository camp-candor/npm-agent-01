  let defaultRulesPath = ontologyDir
    ? path.join(ontologyDir, 'rules.json')
    : path.resolve(
        process.cwd(),
        'packages/001.lore/schemas/ontology/rules.json',
      )

  if (!fs.existsSync(defaultRulesPath)) {
